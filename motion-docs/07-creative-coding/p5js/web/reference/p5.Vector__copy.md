## Example code (extracted)

```js
function setup() {
  createCanvas(100 ,100);

  background(200);

  // Create a p5.Vector object.
  let pos = createVector(50, 50);

  // Make a copy.
  let pc = pos.copy();

  // Draw the point.
  strokeWeight(5);
  point(pc);

  describe('A black point drawn in the middle of a gray square.');
}
```
