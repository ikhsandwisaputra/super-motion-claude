Rotates a 2D vector to a specific angle without changing its magnitude.

By convention, the positive x-axis has an angle of 0. Angles increase in the clockwise direction.

If the vector was created with [createVector()](https://p5js.org/reference/p5/createVector/), `setHeading()` uses the units of the current [angleMode()](https://p5js.org/reference/p5/angleMode/).

## Examples

## Syntax

`setHeading(angle)`

## Parameters

angle

Number: angle of rotation.

function setup() {
  createCanvas(100, 100);

  describe('Two arrows extend from the center of a gray square. The red arrow points to the right and the blue arrow points down.');

function draw() {
  background(200);

  // Create p5.Vector objects.
  let v0 = createVector(50, 50);
  let v1 = createVector(30, 0);

  // Draw the red arrow.
  drawArrow(v0, v1, 'red');

  // Point down.
  v1.setHeading(HALF_PI);

  // Draw the blue arrow.
  drawArrow(v0, v1, 'blue');
}

// Draws an arrow between two vectors.
function drawArrow(base, vec, myColor) {
  push();
  stroke(myColor);
  strokeWeight(3);
  fill(myColor);
  translate(base.x, base.y);
  line(0, 0, vec.x, vec.y);
  rotate(vec.heading());
  let arrowSize = 7;
  translate(vec.mag() - arrowSize, 0);
  triangle(0, arrowSize / 2, 0, -arrowSize / 2, arrowSize, 0);
  pop();
}
