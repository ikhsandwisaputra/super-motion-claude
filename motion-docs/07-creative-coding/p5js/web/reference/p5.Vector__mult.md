Multiplies a vector's components.

`mult()` can use separate numbers, as in `v.mult(1, 2, 3)`, another [p5.Vector](https://p5js.org/reference/p5/p5.Vector) object, as in `v.mult(v2)`, or an array of numbers, as in `v.mult([1, 2, 3])`.

If only one value is provided, as in `v.mult(2)`, then all the components will be multiplied by 2. Calling `mult()` with no arguments, as in `v.mult()`, has no effect.

You should multiply vectors only when they are the same size. When two vectors of different sizes are multiplied, the smaller dimension will be used, any additional values of the longer vector will be ignored. For example, multiplying `[1, 2, 3]` by `[4, 5]` will result in `[4, 10]`.

The static version of `mult()`, as in `p5.Vector.mult(v, 2)`, returns a new [p5.Vector](https://p5js.org/reference/p5/p5.Vector) object and doesn't change the originals.

## Syntax

`mult(n)``mult(x, y, [z])``mult(arr)``mult(v)`

## Parameters

n

Number: The number to multiply with the vector

x

Number: number to multiply with the x component of the vector.

y

Number: number to multiply with the y component of the vector.

z

Number: number to multiply with the z component of the vector.

arr

Number[]: array to multiply with the components of the vector.

v

p5.Vector: vector to multiply with the components of the original vector.

function setup() {
  createCanvas(100, 100);

  background(200);

  // Style the points.
  strokeWeight(5);

  // Top-left.
  let p = createVector(25, 25);
  point(p);

  // Bottom-right.
  // Create a new p5.Vector with
  // p3.x = p.x * p2.x
  // p3.y = p.y * p2.y
  let p2 = createVector(2, 3);
  let p3 = p5.Vector.mult(p, p2);
  point(p3);

  describe('Two black dots drawn on a gray square. One dot is in the top left corner and the other is in the bottom center.');
}

function setup() {
  createCanvas(100, 100);

  describe('Two arrows extending from the top left corner. The blue arrow is twice the length of the red arrow.');
}

function draw() {
  background(200);

  let origin = createVector(0, 0);

  // Draw the red arrow.
  let v1 = createVector(25, 25);
  drawArrow(origin, v1, 'red');

  // Draw the blue arrow.
  let v2 = p5.Vector.mult(v1, 2);
  drawArrow(origin, v2, 'blue');
}

// Draws an arrow between two vectors.
function drawArrow(base, vec, myColor) {
  push();
  stroke(myColor);
  strokeWeight(3);
  fill(myColor);
  translate(base.x, base.y);
  line(0, 0, vec.x, vec.y);
  rotate(vec.heading());
  let arrowSize = 7;
  translate(vec.mag() - arrowSize, 0);
  triangle(0, arrowSize / 2, 0, -arrowSize / 2, arrowSize, 0);
  pop();
}
