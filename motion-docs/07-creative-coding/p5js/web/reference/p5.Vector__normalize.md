Scales the components of a [p5.Vector](https://p5js.org/reference/p5/p5.Vector) object so that its magnitude is 1.

The static version of `normalize()`, as in `p5.Vector.normalize(v)`, returns a new [p5.Vector](https://p5js.org/reference/p5/p5.Vector) object and doesn't change the original.

## Examples

## Syntax

`normalize()`

## Parameters

## Returns

p5.Vector: normalized [p5.Vector](https://p5js.org/reference/p5/p5.Vector/).

function setup() {
  createCanvas(100, 100);

  describe("A red and blue arrow extend from the center of a circle. Both arrows follow the mouse, but the blue arrow's length is fixed to the circle's radius.");
}

function draw() {
  background(240);

  // Vector to the center.
  let v0 = createVector(50, 50);

  // Vector from the center to the mouse.
  let v1 = createVector(mouseX - 50, mouseY - 50);

  // Circle's radius.
  let r = 25;

  // Draw the red arrow.
  drawArrow(v0, v1, 'red');

  // Draw the blue arrow.
  v1.normalize();
  drawArrow(v0, v1.mult(r), 'blue');

  // Draw the circle.
  noFill();
  circle(50, 50, r * 2);
}

// Draws an arrow between two vectors.
function drawArrow(base, vec, myColor) {
  push();
  stroke(myColor);
  strokeWeight(3);
  fill(myColor);
  translate(base.x, base.y);
  line(0, 0, vec.x, vec.y);
  rotate(vec.heading());
  let arrowSize = 7;
  translate(vec.mag() - arrowSize, 0);
  triangle(0, arrowSize / 2, 0, -arrowSize / 2, arrowSize, 0);
  pop();
}
