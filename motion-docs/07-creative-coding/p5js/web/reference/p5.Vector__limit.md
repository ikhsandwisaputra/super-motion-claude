Limits a vector's magnitude to a maximum value.

The static version of `limit()`, as in `p5.Vector.limit(v, 5)`, returns a new [p5.Vector](https://p5js.org/reference/p5/p5.Vector) object and doesn't change the original.

## Examples

## Syntax

`limit(max)`

## Parameters

max

Number: maximum magnitude for the vector.

function setup() {
  createCanvas(100, 100);

  describe("A red and blue arrow extend from the center of a circle. Both arrows follow the mouse, but the blue arrow never crosses the circle's edge.");

function draw() {
  background(240);

  // Vector to the center.
  let v0 = createVector(50, 50);

  // Vector from the center to the mouse.
  let v1 = createVector(mouseX - 50, mouseY - 50);

  // Circle's radius.
  let r = 25;

  // Draw the red arrow.
  drawArrow(v0, v1, 'red');

  // Draw the blue arrow.
  drawArrow(v0, v1.limit(r), 'blue');

  // Draw the circle.
  noFill();
  circle(50, 50, r * 2);
}

// Draws an arrow between two vectors.
function drawArrow(base, vec, myColor) {
  push();
  stroke(myColor);
  strokeWeight(3);
  fill(myColor);
  translate(base.x, base.y);
  line(0, 0, vec.x, vec.y);
  rotate(vec.heading());
  let arrowSize = 7;
  translate(vec.mag() - arrowSize, 0);
  triangle(0, arrowSize / 2, 0, -arrowSize / 2, arrowSize, 0);
  pop();
}
