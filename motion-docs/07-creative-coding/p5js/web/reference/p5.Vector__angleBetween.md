Calculates the angle between two vectors.

The angles returned are signed, which means that `v1.angleBetween(v2) === -v2.angleBetween(v1)`.

If the vector was created with [createVector()](https://p5js.org/reference/p5/createVector/), `angleBetween()` returns angles in the units of the current [angleMode()](https://p5js.org/reference/p5/angleMode/).

## Examples

## Syntax

`angleBetween(value)`

## Parameters

value

p5.Vector: x, y, and z components of a [p5.Vector](https://p5js.org/reference/p5/p5.Vector/).

## Returns

Number: angle between the vectors.

function setup() {
  createCanvas(100, 100);

  describe('Two arrows extend from the center of a gray square. A red arrow points to the right and a blue arrow points down. The text "Radians: 1.57" and "Degrees: 90" is written above the arrows.');
}
function draw() {
  background(200);

  // Create p5.Vector objects.
  let v0 = createVector(50, 50);
  let v1 = createVector(30, 0);
  let v2 = createVector(0, 30);

  // Draw the red arrow.
  drawArrow(v0, v1, 'red');

  // Draw the blue arrow.
  drawArrow(v0, v2, 'blue');

  // Use radians.
  angleMode(RADIANS);

  // Display the angle in radians.
  let angle = round(v1.angleBetween(v2), 2);
  text(`Radians: ${angle}`, 20, 20);

  // Use degrees.
  angleMode(DEGREES);

  // Display the angle in degrees.
  angle = round(v1.angleBetween(v2), 2);
  text(`Degrees: ${angle}`, 20, 35);
}

// Draws an arrow between two vectors.
function drawArrow(base, vec, myColor) {
  push();
  stroke(myColor);
  strokeWeight(3);
  fill(myColor);
  translate(base.x, base.y);
  line(0, 0, vec.x, vec.y);
  rotate(vec.heading());
  let arrowSize = 7;
  translate(vec.mag() - arrowSize, 0);
  triangle(0, arrowSize / 2, 0, -arrowSize / 2, arrowSize, 0);
  pop();
}
